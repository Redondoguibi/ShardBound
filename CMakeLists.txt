cmake_minimum_required(VERSION 3.25)
project(GameRuntime VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The supported baseline is Windows x64 with MSVC.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(game_options INTERFACE)
target_compile_options(game_options INTERFACE /W4 /permissive- /utf-8 /EHsc)
target_compile_definitions(game_options INTERFACE NOMINMAX WIN32_LEAN_AND_MEAN)
add_library(game_core STATIC src/core/Log.cpp src/core/GameLoop.cpp src/core/Math.cpp)
target_include_directories(game_core PUBLIC src)
target_link_libraries(game_core PUBLIC game_options)
add_executable(game src/main.cpp src/core/Application.cpp)
find_package(SDL3 CONFIG REQUIRED)
add_library(game_platform STATIC src/platform/InputSystem.cpp src/platform/windows/WindowSystem.cpp src/platform/windows/TextImage.cpp)
target_link_libraries(game_platform PUBLIC game_core PRIVATE SDL3::SDL3)
target_link_libraries(game_platform PRIVATE gdi32)
add_library(game_renderer STATIC src/renderer/d3d11/Renderer.cpp)
target_link_libraries(game_renderer PUBLIC game_core PRIVATE d3d11 dxgi d3dcompiler)
add_library(game_demo STATIC src/game/DemoScene.cpp src/game/PreparationScreen.cpp)
target_link_libraries(game_demo PUBLIC game_renderer PRIVATE game_platform)
target_link_libraries(game PRIVATE game_platform game_demo)
file(COPY assets/core/shaders DESTINATION "${CMAKE_BINARY_DIR}/assets/core")
find_package(directxtex CONFIG REQUIRED)
find_package(fastgltf CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
add_library(game_assets STATIC src/assets/AssetId.cpp src/assets/VirtualFileSystem.cpp src/assets/Fingerprint.cpp src/assets/TextureCompiler.cpp src/assets/AssetManager.cpp)
target_link_libraries(game_assets PUBLIC game_renderer PRIVATE Microsoft::DirectXTex bcrypt ole32)
target_sources(game_assets PRIVATE src/assets/ModelCompiler.cpp)
target_link_libraries(game_assets PRIVATE fastgltf::fastgltf nlohmann_json::nlohmann_json)
target_link_libraries(game PRIVATE game_assets)
file(COPY assets/core/textures DESTINATION "${CMAKE_BINARY_DIR}/assets/core")
file(COPY assets/core/models DESTINATION "${CMAKE_BINARY_DIR}/assets/core")
find_package(EnTT CONFIG REQUIRED)
add_library(game_ecs STATIC src/ecs/World.cpp src/ecs/RenderSystem.cpp)
target_link_libraries(game_ecs PUBLIC game_assets PRIVATE EnTT::EnTT)
target_link_libraries(game PRIVATE game_ecs)
add_library(game_mods STATIC src/mods/Manifest.cpp src/mods/ModLoader.cpp)
target_link_libraries(game_mods PUBLIC game_assets PRIVATE nlohmann_json::nlohmann_json)
target_link_libraries(game PRIVATE game_mods)
file(COPY mods DESTINATION "${CMAKE_BINARY_DIR}")
add_library(game_scripting STATIC src/scripting/Lexer.cpp src/scripting/Parser.cpp src/scripting/Value.cpp src/scripting/Registry.cpp src/scripting/Runtime.cpp)
target_link_libraries(game_scripting PUBLIC game_assets)
add_library(game_bridge STATIC src/scripting/Bridge.cpp)
target_link_libraries(game_bridge PUBLIC game_scripting game_ecs)
target_link_libraries(game PRIVATE game_bridge)
include(CTest)
if(BUILD_TESTING)
    add_executable(core_tests tests/CoreTests.cpp)
    target_link_libraries(core_tests PRIVATE game_platform)
    add_test(NAME core COMMAND core_tests)
    add_test(NAME runtime_smoke COMMAND game --hidden --frames 3)
    add_executable(renderer_tests tests/RendererTests.cpp)
    target_link_libraries(renderer_tests PRIVATE game_platform game_demo)
    add_test(NAME renderer COMMAND renderer_tests)
    add_executable(asset_tests tests/AssetTests.cpp)
    target_link_libraries(asset_tests PRIVATE game_assets game_platform game_mods)
    add_test(NAME assets COMMAND asset_tests)
    add_executable(world_tests tests/WorldTests.cpp)
    target_link_libraries(world_tests PRIVATE game_ecs)
    add_test(NAME world COMMAND world_tests)
    add_executable(mod_tests tests/ModTests.cpp)
    target_link_libraries(mod_tests PRIVATE game_mods)
    add_test(NAME mods COMMAND mod_tests)
    add_executable(script_tests tests/ScriptTests.cpp)
    target_link_libraries(script_tests PRIVATE game_scripting)
    add_test(NAME scripting COMMAND script_tests)
    add_executable(bridge_tests tests/BridgeTests.cpp)
    target_link_libraries(bridge_tests PRIVATE game_bridge game_mods game_platform)
    add_test(NAME bridge COMMAND bridge_tests)
endif()
