# Preserve the retired default package from pre-procedural builds outside discovery.
# Other user-installed mods in the build directory remain installed.
set(legacy_content "${CMAKE_BINARY_DIR}/content/adventure")
if(NOT EXISTS "${CMAKE_SOURCE_DIR}/content/adventure" AND EXISTS "${CMAKE_SOURCE_DIR}/examples/legacy/adventure/mod.json" AND EXISTS "${legacy_content}/mod.json")
    file(READ "${legacy_content}/mod.json" legacy_manifest)
    string(JSON legacy_id ERROR_VARIABLE legacy_error GET "${legacy_manifest}" id)
    if(NOT legacy_error AND legacy_id STREQUAL "adventure")
        string(TIMESTAMP legacy_stamp "%Y%m%d%H%M%S")
        file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/retired-content")
        file(RENAME "${legacy_content}" "${CMAKE_BINARY_DIR}/retired-content/adventure-${legacy_stamp}" RESULT legacy_move NO_REPLACE)
        if(NOT legacy_move STREQUAL "0")
            message(FATAL_ERROR "Could not preserve retired adventure package: ${legacy_move}")
        endif()
    endif()
endif()

# The embedded Studio uses the same controller and exporter as the Desktop plugin.
set(editor_studio_source "")
foreach(module core templates controller materials validation exporter)
    set(source "${CMAKE_CURRENT_SOURCE_DIR}/BLOCKBENCH/studio/src/${module}.mjs")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
    file(READ "${source}" script)
    string(REGEX REPLACE "(^|\n)import [^\n]*" "\\1" script "${script}")
    string(REGEX REPLACE "(^|\n)export " "\\1" script "${script}")
    string(APPEND editor_studio_source "${script}\n")
endforeach()
