cmake_minimum_required(VERSION 3.25)
project(GameRuntime VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The supported baseline is Windows x64 with MSVC.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(game_options INTERFACE)
target_compile_options(game_options INTERFACE /W4 /permissive- /utf-8 /EHsc)
target_compile_definitions(game_options INTERFACE NOMINMAX WIN32_LEAN_AND_MEAN)
# Bounded AST/import recursion needs headroom with MSVC Debug stack frames.
target_link_options(game_options INTERFACE /STACK:8388608)
add_library(game_core STATIC src/core/Log.cpp src/core/GameLoop.cpp src/core/Math.cpp)
target_include_directories(game_core PUBLIC src)
target_link_libraries(game_core PUBLIC game_options)
add_executable(game src/main.cpp src/core/Application.cpp)
find_package(SDL3 CONFIG REQUIRED)
add_library(game_platform STATIC src/platform/InputSystem.cpp src/platform/windows/WindowSystem.cpp src/platform/windows/TextImage.cpp)
target_link_libraries(game_platform PUBLIC game_core PRIVATE SDL3::SDL3)
target_link_libraries(game_platform PRIVATE gdi32)
add_library(game_renderer STATIC src/renderer/d3d11/Renderer.cpp)
target_link_libraries(game_renderer PUBLIC game_core PRIVATE d3d11 dxgi d3dcompiler)
add_library(game_demo STATIC src/game/DemoScene.cpp src/game/PreparationScreen.cpp)
target_link_libraries(game_demo PUBLIC game_renderer PRIVATE game_platform)
target_link_libraries(game PRIVATE game_platform game_demo)
file(COPY assets/core/shaders DESTINATION "${CMAKE_BINARY_DIR}/assets/core")
find_package(directxtex CONFIG REQUIRED)
find_package(fastgltf CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
target_sources(game_core PRIVATE src/core/Data.cpp)
target_link_libraries(game_core PRIVATE nlohmann_json::nlohmann_json)
add_library(game_assets STATIC src/assets/AssetId.cpp src/assets/VirtualFileSystem.cpp src/assets/Fingerprint.cpp src/assets/TextureCompiler.cpp src/assets/AssetManager.cpp)
target_link_libraries(game_assets PUBLIC game_renderer PRIVATE Microsoft::DirectXTex bcrypt ole32)
target_sources(game_assets PRIVATE src/assets/ModelCompiler.cpp src/assets/CoreAssets.cpp)
target_link_libraries(game_assets PRIVATE fastgltf::fastgltf nlohmann_json::nlohmann_json)
target_link_libraries(game PRIVATE game_assets)
file(COPY assets/core/textures DESTINATION "${CMAKE_BINARY_DIR}/assets/core")
file(COPY assets/core/models DESTINATION "${CMAKE_BINARY_DIR}/assets/core")
find_package(EnTT CONFIG REQUIRED)
add_library(game_ecs STATIC src/ecs/World.cpp src/ecs/RenderSystem.cpp)
target_link_libraries(game_ecs PUBLIC game_assets PRIVATE EnTT::EnTT)
target_link_libraries(game_ecs PRIVATE ole32)
target_link_libraries(game PRIVATE game_ecs)
add_library(game_mods STATIC src/mods/Manifest.cpp src/mods/ModLoader.cpp)
target_link_libraries(game_mods PUBLIC game_assets PRIVATE nlohmann_json::nlohmann_json)
target_link_libraries(game PRIVATE game_mods)
file(COPY mods DESTINATION "${CMAKE_BINARY_DIR}")
add_library(game_scripting STATIC src/scripting/Lexer.cpp src/scripting/Parser.cpp src/scripting/Value.cpp src/scripting/Registry.cpp src/scripting/Runtime.cpp)
target_link_libraries(game_scripting PUBLIC game_assets)
add_library(game_bridge STATIC src/scripting/Bridge.cpp)
target_link_libraries(game_bridge PUBLIC game_scripting game_ecs)
target_link_libraries(game PRIVATE game_bridge)
find_package(Jolt CONFIG REQUIRED)
find_path(MINIAUDIO_INCLUDE_DIR miniaudio.h REQUIRED)
add_library(game_physics STATIC src/physics/PhysicsWorld.cpp)
target_link_libraries(game_physics PUBLIC game_core PRIVATE Jolt::Jolt)
add_library(game_audio STATIC src/audio/AudioSystem.cpp)
target_include_directories(game_audio SYSTEM PRIVATE "${MINIAUDIO_INCLUDE_DIR}")
target_link_libraries(game_audio PUBLIC game_core)
file(COPY assets/core/audio DESTINATION "${CMAKE_BINARY_DIR}/assets/core")
add_library(game_gameplay STATIC src/game/HealthSystem.cpp src/game/InventorySystem.cpp src/game/InteractionSystem.cpp src/game/PhysicsSystem.cpp src/game/PlayerController.cpp)
target_link_libraries(game_gameplay PUBLIC game_ecs game_physics)
add_library(game_gameplay_bridge STATIC src/scripting/GameplayBridge.cpp)
target_link_libraries(game_gameplay_bridge PUBLIC game_bridge game_gameplay game_audio game_platform)
target_link_libraries(game PRIVATE game_gameplay_bridge)
add_library(game_ui STATIC src/game/GameUi.cpp)
target_link_libraries(game_ui PUBLIC game_bridge game_platform)
target_link_libraries(game PRIVATE game_ui)
file(COPY examples DESTINATION "${CMAKE_BINARY_DIR}")
file(COPY content DESTINATION "${CMAKE_BINARY_DIR}")
target_sources(game_scripting PRIVATE src/scripting/ScriptCheck.cpp)
target_sources(game_scripting PRIVATE src/scripting/DataConversion.cpp)
add_library(game_extensions STATIC src/extensions/Components.cpp)
target_link_libraries(game_extensions PUBLIC game_ecs)
add_library(game_saves STATIC src/persistence/SaveGame.cpp)
target_link_libraries(game_saves PUBLIC game_gameplay PRIVATE nlohmann_json::nlohmann_json)
add_library(game_native STATIC src/extensions/NativeMods.cpp)
target_include_directories(game_native PRIVATE sdk/native)
target_link_libraries(game_native PUBLIC game_extensions game_saves game_bridge PRIVATE nlohmann_json::nlohmann_json)
add_library(game_extension_bridge STATIC src/scripting/ExtensionBridge.cpp)
target_link_libraries(game_extension_bridge PUBLIC game_extensions game_saves game_bridge)
target_link_libraries(game PRIVATE game_native game_extension_bridge)
add_library(native_sample SHARED examples/native/sample.cpp)
target_include_directories(native_sample PRIVATE sdk/native)
target_link_libraries(native_sample PRIVATE game_options)
set_target_properties(native_sample PROPERTIES OUTPUT_NAME sample RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/examples/native/native_sample")
add_executable(modcheck src/tools/ModCheck.cpp)
target_link_libraries(modcheck PRIVATE game_scripting game_mods nlohmann_json::nlohmann_json)
target_link_libraries(modcheck PRIVATE game_saves)
