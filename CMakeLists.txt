cmake_minimum_required(VERSION 3.25)
project(GameRuntime VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The supported baseline is Windows x64 with MSVC.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(game_options INTERFACE)
target_compile_options(game_options INTERFACE /W4 /permissive- /utf-8 /EHsc)
target_compile_definitions(game_options INTERFACE NOMINMAX WIN32_LEAN_AND_MEAN)
add_library(game_core STATIC src/core/Log.cpp src/core/GameLoop.cpp)
target_include_directories(game_core PUBLIC src)
target_link_libraries(game_core PUBLIC game_options)
add_executable(game src/main.cpp src/core/Application.cpp)
target_link_libraries(game PRIVATE game_core)
include(CTest)
if(BUILD_TESTING)
    add_executable(core_tests tests/CoreTests.cpp)
    target_link_libraries(core_tests PRIVATE game_core)
    add_test(NAME core COMMAND core_tests)
    add_test(NAME runtime_smoke COMMAND game --frames 3)
endif()
